chocolate_filtered <- chocolate %>%
  group_by(country_of_bean_origin) %>%
  filter(n() >= 50) %>%
  ungroup()

chocolate_filtered <- chocolate_filtered %>%
  mutate(
    cocoa_percent_num = as.numeric(substr(cocoa_percent, 1, 2)),
    cocoa_group = case_when(
      cocoa_percent_num < 60 ~ "<60%",
      cocoa_percent_num >= 60 & cocoa_percent_num < 70 ~ "60-<70%",
      cocoa_percent_num >= 70 & cocoa_percent_num < 90 ~ "70-<90%",
      cocoa_percent_num >= 90 ~ ">=90%"
    )
  )

chocolate_filtered <- chocolate_filtered %>%
  mutate(
    cocoa_group = fct_relevel(
      cocoa_group,
      "<60%",
      "60-<70%",
      "70-<90%",
      ">=90%"
    )
  )

ggplot(chocolate_filtered, aes(x = cocoa_group, y = rating)) +
  geom_boxplot() +
  facet_wrap(~ country_of_bean_origin) +
  labs(
    title = "Chocolate Ratings by Cocoa Percentage and Country",
    x = "Chocolate Percentage",
    y = "Rating"
  )
